     // 用户信息
     message User {
       int64 id = 1;                    // 用户ID
       string username = 2;              // 用户名
       string avatar_url = 3;            // 头像URL
       string nickname = 4;               // 昵称
       string created_at = 5;             // 创建时间
       string updated_at = 6;             // 更新时间
     }

     // 视频信息
     message Video {
       int64 id = 1;                    // 视频ID
       int64 user_id = 2;                // 作者ID
       string title = 3;                  // 视频标题
       string description = 4;          // 视频描述
       string video_url = 5;            // 视频URL
       int64 views = 6;                   // 播放量
       int64 likes = 7;                   // 点赞数
       int64 comments = 8;                // 评论数
       string created_at = 9;             // 创建时间
       string updated_at = 10;            // 更新时间
       User author = 11;                 // 作者信息
     }

     // 评论信息
     message Comment {
       int64 id = 1;                    // 评论ID
       int64 user_id = 2;                // 评论用户ID
       int64 video_id = 3;              // 视频ID
       int64 parent_id = 4;              // 父评论ID（0表示一级评论，否则为回复）
       string content = 5;                // 评论内容
       int64 likes = 6;                   // 点赞数
       string created_at = 7;             // 创建时间
       string updated_at = 8;            // 更新时间
       User user = 9;                    // 评论用户信息
       repeated Comment replies = 10;         // 回复列表
     }

     // 点赞类型枚举
     enum LikeType {
       LIKE_TYPE_VIDEO = 1;    // 视频点赞
       LIKE_TYPE_COMMENT = 2;    // 评论点赞
     }

     // 关注关系信息
     message Follow {
       int64 id = 1;                    // 关注记录ID
       int64 follower_id = 2;           // 关注者ID
       int64 following_id = 3;         // 被关注者ID
       string created_at = 4;             // 创建时间
       User follower = 5;                // 关注者信息
       User following = 6;              // 被关注者信息
     }

     // 用户列表响应
     message UserListResponse {
       repeated User users = 1;           // 用户列表
       int32 total = 2;                    // 总数
     }

     // 视频列表响应
     message VideoListResponse {
       repeated Video videos = 1;         // 视频列表
       int32 total = 2;                    // 总数
     }

     // 评论列表响应
     message CommentListResponse {
       repeated Comment comments = 1;     // 评论列表
       int32 total = 2;                    // 总数
     }
